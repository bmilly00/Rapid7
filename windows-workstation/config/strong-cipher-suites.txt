# TLS cipher-suite order applied by scripts\03-Harden-TLS.ps1
#
# Registry : HKLM\SOFTWARE\Policies\Microsoft\Cryptography\Configuration\SSL\00010002
# Value    : Functions (REG_SZ) - ONE line, comma-separated, no spaces
# GPO path : Computer Configuration > Administrative Templates > Network
#            > SSL Configuration Settings > SSL Cipher Suite Order
#
# TLS 1.3 suites first (used on Windows 11 / Server 2022+; older builds
# ignore unknown names), then TLS 1.2 ECDHE + AES-GCM only:
#   - no TLS_RSA_* (static key exchange)          -> fixes "static key ciphers"
#   - no 3DES                                     -> fixes "3DES cipher suite" (SWEET32)
#   - no MD5/SHA-1 MACs (AEAD or SHA-2 only)      -> fixes "weak MAC cipher suites"
#   - AES-GCM / TLS 1.3 AEAD suites present       -> fixes "no strong cipher algorithms"

TLS_AES_256_GCM_SHA384,TLS_AES_128_GCM_SHA256,TLS_ECDHE_ECDSA_WITH_AES_256_GCM_SHA384,TLS_ECDHE_ECDSA_WITH_AES_128_GCM_SHA256,TLS_ECDHE_RSA_WITH_AES_256_GCM_SHA384,TLS_ECDHE_RSA_WITH_AES_128_GCM_SHA256
